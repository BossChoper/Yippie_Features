// Function to extract Google reviews from the current page
function scrapeGoogleReviews(maxReviews = 5) {
  const reviews = [];
  
  // Wait for reviews to load (you may need to manually scroll to load more reviews)
  setTimeout(() => {
    // Get all review elements
    const reviewElements = document.querySelectorAll('.jftiEf');
    
    // Extract data from each review
    reviewElements.forEach((review, index) => {
      if (index >= maxReviews) return;
      
      try {
        const author = review.querySelector('.d4r55')?.textContent.trim() || 'Unknown';
        const rating = review.querySelector('.Fam1ne')?.getAttribute('aria-label')?.match(/\d+/)?.[0] || '0';
        const text = review.querySelector('.wiI7pd')?.textContent.trim() || '';
        const date = review.querySelector('.rsqaWe')?.textContent.trim() || '';
        
        reviews.push({
          author,
          rating: parseInt(rating),
          text,
          date
        });
      } catch (error) {
        console.error('Error parsing review:', error);
      }
    });
    
    console.log('Scraped Reviews:', reviews);
    return reviews;
  }, 3000); // Wait 3 seconds for reviews to load
}

// Usage:
// 1. Go to Google Maps and search for your restaurant
// 2. Open the reviews section
// 3. Scroll down to load more reviews
// 4. Open browser console (F12) and paste this code
// 5. Call the function: scrapeGoogleReviews(10) // Change number to get more/less reviews

// Optional: Copy results to clipboard
function copyReviewsToClipboard() {
  const reviews = scrapeGoogleReviews();
  navigator.clipboard.writeText(JSON.stringify(reviews, null, 2))
    .then(() => console.log('Reviews copied to clipboard!'))
    .catch(err => console.error('Failed to copy:', err));
}
